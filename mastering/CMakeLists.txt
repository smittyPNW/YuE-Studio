cmake_minimum_required(VERSION 3.22)
project(YuEStudioMastering VERSION 1.0 LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0")
set(CMAKE_OSX_ARCHITECTURES "arm64")
set(JUCE_ROOT "" CACHE PATH "Optional local JUCE source checkout")
if(JUCE_ROOT)
  add_subdirectory("${JUCE_ROOT}" juce)
else()
  include(FetchContent)
  FetchContent_Declare(JUCE
    GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG d6181bde38d858c283c3b7bf699ce6340c050b5d
    GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(JUCE)
endif()
juce_add_console_app(StudioMasterEngine PRODUCT_NAME "StudioMasterEngine")
juce_generate_juce_header(StudioMasterEngine)
file(GLOB DSP CONFIGURE_DEPENDS vendor/dsp/*.cpp)
file(GLOB AUDIO CONFIGURE_DEPENDS vendor/audio/*.cpp)
target_sources(StudioMasterEngine PRIVATE Source/Main.cpp ${DSP} ${AUDIO} vendor/presets/PresetManager.cpp)
target_include_directories(StudioMasterEngine PRIVATE vendor)
target_compile_definitions(StudioMasterEngine PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0 JUCE_USE_FLAC=1 JUCE_USE_MP3AUDIOFORMAT=1)
target_link_libraries(StudioMasterEngine PRIVATE juce::juce_audio_formats juce::juce_dsp juce::juce_recommended_config_flags)
